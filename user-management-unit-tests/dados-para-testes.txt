// UserService.create(UserCreateDTO data)
// Passam pelas validacoes de campos, tamanho do nome, formato do e-mail e tamanho da senha:
UserCreateDTO createValido1 = new UserCreateDTO("Maria Silva", "maria.silva@example.com", "senha123");
UserCreateDTO createValido2 = new UserCreateDTO("Joao", "joao@exemplo.com", "abc123");

// Nao passam pelas validacoes:
UserCreateDTO createInvalido1 = new UserCreateDTO("Ana", "ana@example.com", "senha123");
// Nome com menos de 4 caracteres.
UserCreateDTO createInvalido2 = new UserCreateDTO("Carlos", "carlos-email-invalido", "123");
// E-mail fora do formato esperado e senha com menos de 6 caracteres.
